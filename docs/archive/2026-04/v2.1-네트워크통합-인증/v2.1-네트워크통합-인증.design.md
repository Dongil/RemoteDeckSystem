### 2.3 network/NetManager.h (REWRITE)

```cpp
#pragma once
#include <Arduino.h>
#include <string>
#include <functional>
#include <IPAddress.h>
#include <WiFi.h>
#include <ETH.h>
#include <SPI.h>
#include "config/DeviceConfig.h"
#include "config/PinConfig.h"

enum class NetMode { NONE, ETHERNET, WIFI_STA };

class NetManager {
public:
    void begin(NetworkConfig& config);
    void loop();

    bool isConnected() const;
    NetMode activeMode() const { return _mode; }
    IPAddress localIP() const;
    std::string macAddress() const;

    // For MQTT (both ETH and WiFi use WiFiClient via lwIP)
    WiFiClient& getNetClient() { return _netClient; }

    // Callback when network becomes available
    using ConnectedCallback = std::function<void()>;
    void setOnConnected(ConnectedCallback cb) { _onConnected = cb; }

private:
    NetworkConfig* _config = nullptr;
    NetMode _mode = NetMode::NONE;
    WiFiClient _netClient;
    bool _connected = false;
    bool _notifiedConnected = false;

    ConnectedCallback _onConnected = nullptr;

    void initEthernet(NetworkConfig& config);
    void initWiFiSTA(NetworkConfig& config);
    static IPAddress strToIP(const std::string& str);

    static void onNetworkEvent(WiFiEvent_t event);
    static NetManager* _instance;
};
```

### 2.4 network/NetManager.cpp (REWRITE)

**핵심 로직:**

```cpp
void NetManager::begin(NetworkConfig& config) {
    _config = &config;
    _instance = this;
    WiFi.onEvent(onNetworkEvent);

    if (config.mode == "ethernet" || config.mode == "") {
        initEthernet(config);
    } else if (config.mode == "wifi") {
        initWiFiSTA(config);
    }
}

void NetManager::initEthernet(NetworkConfig& config) {
    SPI.begin(PIN_ETH_SCK, PIN_ETH_MISO, PIN_ETH_MOSI, PIN_ETH_CS);

    // ESP32 Arduino 표준 방식
    if (!ETH.begin(ETH_PHY_W5500, 1, PIN_ETH_CS, PIN_ETH_INT, -1, SPI)) {
        Serial.println("Network: ETH.begin() failed");
        return;
    }
    // Static IP는 ETH_CONNECTED 이벤트에서 적용
    // 비동기 - 블로킹 없음
}

void NetManager::initWiFiSTA(NetworkConfig& config) {
    WiFi.mode(WIFI_STA);
    WiFi.begin(config.wifiSsid.c_str(), config.wifiPassword.c_str());

    // Static IP (DHCP가 아닌 경우)
    if (!config.wifiDhcp && !config.wifiIp.empty()) {
        WiFi.config(strToIP(config.wifiIp),
                    strToIP(config.wifiGateway),
                    strToIP(config.wifiSubnet),
                    strToIP(config.wifiDns1));
    }
    // 비동기 - 연결 완료는 이벤트로 통지
}

void NetManager::onNetworkEvent(WiFiEvent_t event) {
    switch (event) {
        case ARDUINO_EVENT_ETH_CONNECTED:
            // Static IP 적용 (ETH)
            if (_instance->_config && !_instance->_config->ethDhcp) {
                ETH.config(strToIP(_instance->_config->ethIp), ...);
            }
            break;
        case ARDUINO_EVENT_ETH_GOT_IP:
            _instance->_connected = true;
            _instance->_mode = NetMode::ETHERNET;
            break;
        case ARDUINO_EVENT_WIFI_STA_GOT_IP:
            _instance->_connected = true;
            _instance->_mode = NetMode::WIFI_STA;
            break;
        case ARDUINO_EVENT_ETH_DISCONNECTED:
        case ARDUINO_EVENT_WIFI_STA_DISCONNECTED:
            _instance->_connected = false;
            break;
    }
}

void NetManager::loop() {
    // 연결 완료 시 콜백 (1회)
    if (_connected && !_notifiedConnected) {
        _notifiedConnected = true;
        digitalWrite(PIN_STATUS1, HIGH);
        if (_onConnected) _onConnected();
    }
    // 연결 끊김 감지
    if (!_connected && _notifiedConnected) {
        _notifiedConnected = false;
        digitalWrite(PIN_STATUS1, LOW);
    }
}

IPAddress NetManager::localIP() const {
    if (_mode == NetMode::ETHERNET) return ETH.localIP();
    if (_mode == NetMode::WIFI_STA) return WiFi.localIP();
    return IPAddress(0, 0, 0, 0);
}
```

```cpp
bool WebServer::requireAuth(AsyncWebServerRequest* req) {
    if (!_auth) return true;
    if (!req->authenticate(_auth->user.c_str(), _auth->pass.c_str())) {
        req->requestAuthentication("RemoteDeck");
        return false;
    }
    return true;
}
```

### 2.6 POST /api/auth (패스워드 변경 API)

| Method | Endpoint | Auth | Request | Response |
|--------|----------|------|---------|----------|
| POST | `/api/auth` | Basic Auth | `{"current_pass":"12345","new_pass":"newpw","confirm_pass":"newpw"}` | `{"ok":true}` or `{"ok":false,"error":"..."}` |

**검증 로직:**
1. Basic Auth 헤더 검증
2. `current_pass`가 현재 저장된 PW와 일치 확인
3. `new_pass`와 `confirm_pass` 일치 확인
4. 새 PW 저장 (ConfigManager)

### 4.2 app.js - 인증 관련

```javascript
// 설정 저장 시 브라우저 Basic Auth 자동 처리
// (401 → 브라우저 ID/PW 다이얼로그 → 재요청)

function changePassword() {
    const curPw = document.getElementById('cur-pw').value;
    const newPw = document.getElementById('new-pw').value;
    const confirmPw = document.getElementById('confirm-pw').value;

    if (newPw !== confirmPw) { alert('패스워드가 일치하지 않습니다'); return; }

    fetch('/api/auth', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({current_pass: curPw, new_pass: newPw, confirm_pass: confirmPw})
    }).then(r => {
        if (r.status === 401) { alert('현재 패스워드가 올바르지 않습니다'); return; }
        return r.json();
    }).then(d => {
        if (d?.ok) alert('패스워드가 변경되었습니다');
    });
}
```
